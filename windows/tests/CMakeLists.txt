cmake_minimum_required(VERSION 3.22)
project(naku_windows_tests LANGUAGES CXX)
add_executable(naku_secure_bridge_test naku_secure_bridge_test.cpp ../runner/naku_secure_bridge.cpp)
target_compile_features(naku_secure_bridge_test PRIVATE cxx_std_17)
target_compile_options(naku_secure_bridge_test PRIVATE /W4 /WX /utf-8 /EHsc)
target_compile_definitions(naku_secure_bridge_test PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(naku_secure_bridge_test PRIVATE bcrypt crypt32 shell32 ole32)
enable_testing()
add_test(NAME windows_nas_crypto COMMAND naku_secure_bridge_test "${CMAKE_CURRENT_BINARY_DIR}/test-credentials.dpapi")
